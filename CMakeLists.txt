cmake_minimum_required(VERSION 3.24)
project(rawengine VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
option(RAWENGINE_SANITIZE "Enable address/undefined sanitizers" OFF)
option(RAWENGINE_STRICT_DEPS "Require versions from the development lock" ON)
find_package(PkgConfig REQUIRED)
pkg_check_modules(RAW REQUIRED IMPORTED_TARGET libraw>=0.21)
pkg_check_modules(LCMS REQUIRED IMPORTED_TARGET lcms2>=2.14)
pkg_check_modules(PNG REQUIRED IMPORTED_TARGET libpng>=1.6)
pkg_check_modules(TIFF REQUIRED IMPORTED_TARGET libtiff-4>=4.5)
pkg_check_modules(JPEG REQUIRED IMPORTED_TARGET libjpeg>=2.1)
pkg_check_modules(GLIB REQUIRED IMPORTED_TARGET glib-2.0>=2.40)
if(RAWENGINE_STRICT_DEPS)
  foreach(pair RAW:0.22.2 LCMS:2.19 PNG:1.6.59 TIFF:4.7.2 JPEG:3.2.0 GLIB:2.90.1)
    string(REPLACE ":" ";" parts "${pair}")
    list(GET parts 0 dep)
    list(GET parts 1 ver)
    if(NOT "${${dep}_VERSION}" STREQUAL "${ver}")
      message(FATAL_ERROR "${dep} must be ${ver}, got ${${dep}_VERSION}")
    endif()
  endforeach()
endif()
if(RAWENGINE_SANITIZE)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
endif()
# Build the unmodified pinned Lensfun sources without its root-project CMake assumptions.
set(VERSION_MAJOR 0)
set(VERSION_MINOR 3)
set(VERSION_MICRO 4)
set(VERSION_BUGFIX 0)
set(LENSFUN_DB_VERSION 1)
set(CMAKE_INSTALL_FULL_DATAROOTDIR "${CMAKE_INSTALL_PREFIX}/share")
set(CMAKE_INSTALL_LOCALSTATEDIR var)
set(LENSFUN_GLIB_REQUIREMENT_MACRO GLIB_VERSION_2_40)
include(CheckIncludeFile)
check_include_file(endian.h HAVE_ENDIAN_H)
configure_file(third_party/lensfun/include/lensfun/config.h.in.cmake generated/config.h)
configure_file(third_party/lensfun/include/lensfun/lensfun.h.in generated/lensfun.h @ONLY)
file(GLOB LENSFUN_SOURCES CONFIGURE_DEPENDS third_party/lensfun/libs/lensfun/*.cpp)
add_library(rawengine_lensfun SHARED ${LENSFUN_SOURCES})
target_include_directories(rawengine_lensfun PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
target_link_libraries(rawengine_lensfun PRIVATE PkgConfig::GLIB)
add_library(rawengine SHARED src/core.cpp src/recipe.cpp src/decode.cpp src/pipeline.cpp src/profile.cpp src/io.cpp src/c_api.cpp)
target_include_directories(rawengine PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include> PRIVATE third_party "${CMAKE_CURRENT_BINARY_DIR}/generated")
target_compile_definitions(rawengine PRIVATE RAWENGINE_LENS_DB="${CMAKE_CURRENT_SOURCE_DIR}/third_party/lensfun/data/db" RAWENGINE_INSTALL_DB="${CMAKE_INSTALL_PREFIX}/share/rawengine/lensfun")
target_compile_options(rawengine PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(rawengine PRIVATE PkgConfig::RAW PkgConfig::LCMS PkgConfig::PNG PkgConfig::TIFF PkgConfig::JPEG rawengine_lensfun)
set_target_properties(rawengine PROPERTIES VERSION ${PROJECT_VERSION} SOVERSION 0)
if(APPLE)
  set_target_properties(rawengine rawengine_lensfun PROPERTIES INSTALL_RPATH "@loader_path")
else()
  set_target_properties(rawengine rawengine_lensfun PROPERTIES INSTALL_RPATH "$ORIGIN")
endif()
add_executable(rawengine_cli src/cli.cpp)
set_target_properties(rawengine_cli PROPERTIES OUTPUT_NAME rawengine)
target_include_directories(rawengine_cli PRIVATE third_party)
target_link_libraries(rawengine_cli PRIVATE rawengine)
if(APPLE)
  set_target_properties(rawengine_cli PROPERTIES INSTALL_RPATH "@loader_path/../lib")
else()
  set_target_properties(rawengine_cli PROPERTIES INSTALL_RPATH "$ORIGIN/../lib")
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(rawengine_tests tests/tests.cpp)
  target_include_directories(rawengine_tests PRIVATE third_party src)
  target_link_libraries(rawengine_tests PRIVATE rawengine PkgConfig::TIFF)
  add_test(NAME rawengine_tests COMMAND rawengine_tests)
  add_executable(rawengine_c_smoke tests/c_smoke.c)
  target_link_libraries(rawengine_c_smoke PRIVATE rawengine)
  add_test(NAME rawengine_c_smoke COMMAND rawengine_c_smoke)
endif()
install(TARGETS rawengine rawengine_lensfun rawengine_cli LIBRARY DESTINATION lib RUNTIME DESTINATION bin)
install(DIRECTORY include/rawengine DESTINATION include)
install(DIRECTORY third_party/lensfun/data/db/ DESTINATION share/rawengine/lensfun FILES_MATCHING PATTERN "*.xml")
install(FILES THIRD_PARTY_NOTICES.md dependencies.lock.json DESTINATION share/rawengine)
install(DIRECTORY third_party/lensfun/docs/ DESTINATION share/rawengine/licenses/lensfun FILES_MATCHING PATTERN "*.txt")
install(FILES third_party/nlohmann/json.hpp DESTINATION share/rawengine/licenses/nlohmann)
if(BUILD_TESTING)
  find_package(Python3 COMPONENTS Interpreter QUIET)
  if(Python3_FOUND)
    add_test(NAME rawengine_color_error COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_color_error.py)
  endif()
endif()
